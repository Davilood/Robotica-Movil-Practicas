import { getCollection } from 'astro:content';
import { CardGrid, LinkCard } from '@astrojs/starlight/components';

export const posts = (await getCollection('docs', ({ id }) => id.startsWith('posts/')))
	.sort((first, second) => second.id.localeCompare(first.id));

## Cuaderno de prácticas

Registro de objetivos, pruebas y resultados del trabajo con el robot móvil.

{posts.length > 0 ? (
	<CardGrid>
		{posts.map((post) => (
			<LinkCard
				title={post.data.title}
				description={post.data.description}
				href={`./${post.id}/`}
			/>
		))}
	</CardGrid>
) : (
	<p>Aún no hay entradas. La primera práctica aparecerá aquí.</p>
)}
